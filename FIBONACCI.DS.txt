To create Fibonacci series using recursive function.

#include<stdio.h>
#include<conio.h>
int fibonacci(int);
void main()
{
int i,n;
clrscr();
printf("\n Enter the no of elements to be displayed:");
scanf("%d",&n);
for(i=0;i<n;i++)
printf("%d\t",fibonacci(i));
getch();
}
int fibonacci(int n)
{
if(n<=0)
return 0;
else if(n==1)
return 1;
else
return fibonacci(n-1)+fibonacci(n-2);
}
